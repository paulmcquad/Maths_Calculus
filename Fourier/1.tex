\section*{Table of Fourier Transforms and Properties}
Let $f(t)$ be the time-domain signal and $F(\omega) = \mathcal{F}\{f(t)\} = \int_{-\infty}^{\infty} f(t) e^{-j\omega t} dt$ be its Fourier transform.

\subsection*{Common Fourier Transform Pairs}
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
\textbf{Function} $f(t)$ & \textbf{Fourier Transform} $F(\omega)$ & \textbf{Condition} \\ \hline
$\delta(t)$ (Dirac delta) & $1$ & --- \\ \hline
$1$ (Constant) & $2\pi\delta(\omega)$ & --- \\ \hline
$e^{-at} u(t)$ (Causal exponential) & $\frac{1}{a + j\omega}$ & $a > 0$ \\ \hline
$e^{-|at|}$ (Double-sided exponential) & $\frac{2a}{a^2 + \omega^2}$ & $a > 0$ \\ \hline
$t e^{-at} u(t)$ & $\frac{1}{(a + j\omega)^2}$ & $a > 0$ \\ \hline
$e^{-at^2}$ (Gaussian) & $\sqrt{\frac{\pi}{a}} e^{-\omega^2 / (4a)}$ & $a > 0$ \\ \hline
$\Pi(t/tau)$ (Rectangular pulse) & $\tau \text{sinc}\left(\frac{\omega\tau}{2}\right)$ & --- \\ \hline
$\cos(\omega_0 t)$ & $\pi [\delta(\omega - \omega_0) + \delta(\omega + \omega_0)]$ & --- \\ \hline
$\sin(\omega_0 t)$ & $j\pi [\delta(\omega + \omega_0) - \delta(\omega - \omega_0)]$ & --- \\ \hline
\end{tabular}
\end{center}

\subsection*{Basic Fourier Transform Properties}
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
\textbf{Property} & \textbf{Time Domain} $f(t)$ & \textbf{Frequency Domain} $F(\omega)$ \\ \hline
Linearity & $a f(t) + b g(t)$ & $a F(\omega) + b G(\omega)$ \\ \hline
Time Shifting & $f(t - t_0)$ & $F(\omega) e^{-j\omega t_0}$ \\ \hline
Frequency Shifting & $f(t) e^{j\omega_0 t}$ & $F(\omega - \omega_0)$ \\ \hline
Time Scaling & $f(at)$ & $\frac{1}{|a|} F\left(\frac{\omega}{a}\right)$ \\ \hline
Time Differentiation & $\frac{d^n}{dt^n} f(t)$ & $(j\omega)^n F(\omega)$ \\ \hline
Time Integration & $\int_{-\infty}^{t} f(\tau) d\tau$ & $\frac{1}{j\omega}F(\omega) + \pi F(0)\delta(\omega)$ \\ \hline
Convolution & $f(t) * g(t)$ & $F(\omega) \cdot G(\omega)$ \\ \hline
Multiplication & $f(t) \cdot g(t)$ & $\frac{1}{2\pi} [F(\omega) * G(\omega)]$ \\ \hline
Duality & $F(t)$ & $2\pi f(-\omega)$ \\ \hline
\end{tabular}
\end{center}
