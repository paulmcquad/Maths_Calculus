				\sectiontitle{General Formulas}

				\begin{tabular}{ll}
					
					\textbf{Rules} & \textbf{Integrals} \\
					
					Constant rule & $\int a \, dx = ax + C$ \\
					Power rule & $\int x^n \, dx = \frac{x^{n+1}}{n+1} + C$ \\
					%Linearity rule & $\frac{d}{dx}[c \cdot f(x)] = c \cdot f'(x)$ \\
					Sum / Difference Rule & $\int (f(x) \pm g(x)) \, dx = \int f(x) \, dx \pm \int g(x) \, dx$ \\
					Product Rule & $\int u \, dv = uv - \int v \, du$ \\
					Quotient Rule &  $\int \frac{f'(x)}{f(x)} \, dx = \ln|f(x)| + C$ \\
					Chain Rule & $\int f(g(x)) \cdot g'(x) \, dx = F(g(x)) + C$ \\					
					Reciprocal Rule & $\int \frac{1}{x} \, dx = \ln|x| + C$ \\
					

					
				\end{tabular}
